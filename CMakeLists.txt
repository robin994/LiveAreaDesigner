cmake_minimum_required(VERSION 3.16)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  if(DEFINED ENV{VITASDK})
    set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "Vita toolchain")
  else()
    set(CMAKE_TOOLCHAIN_FILE "/usr/local/vitasdk/share/vita.toolchain.cmake" CACHE PATH "Vita toolchain")
  endif()
endif()
project(LiveAreaDesigner C)
if(DEFINED ENV{VITASDK})
  set(VITA_SDK_ROOT "$ENV{VITASDK}")
else()
  set(VITA_SDK_ROOT "/usr/local/vitasdk")
endif()
include("${VITA_SDK_ROOT}/share/vita.cmake" REQUIRED)
find_package(SDL2 REQUIRED)
add_executable(livearea_designer src/designer.c src/assets.c src/font.c)
target_compile_features(livearea_designer PRIVATE c_std_99)
target_include_directories(livearea_designer PRIVATE ${SDL2_INCLUDE_DIRS})
target_compile_options(livearea_designer PRIVATE -Wall -Wextra -Werror=implicit-function-declaration)
target_link_libraries(livearea_designer SDL2_image ${SDL2_LIBRARIES} SDL2main png jpeg webpdemux webp z pthread vita2d SceDisplay_stub SceGxm_stub SceCtrl_stub SceTouch_stub SceAudio_stub SceSysmodule_stub SceKernelThreadMgr_stub SceAppMgr_stub m)
vita_create_self(eboot.bin livearea_designer)
vita_create_vpk(LIVAREA01.vpk LIVAREA01 eboot.bin VERSION 01.11 NAME "LiveArea Designer")
